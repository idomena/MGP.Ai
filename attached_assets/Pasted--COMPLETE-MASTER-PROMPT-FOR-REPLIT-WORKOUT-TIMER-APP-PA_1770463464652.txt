```javascript
// Attach drag event listeners
function attachDragListeners() {
  const items = document.querySelectorAll('.exercise-item');
  
  items.forEach((item, index) => {
    // Desktop drag events
    item.addEventListener('dragstart', handleDragStart);
    item.addEventListener('dragover', handleDragOver);
    item.addEventListener('drop', handleDrop);
    item.addEventListener('dragend', handleDragEnd);
    item.addEventListener('dragenter', handleDragEnter);
    item.addEventListener('dragleave', handleDragLeave);
    
    // Mobile touch events for drag
    item.addEventListener('touchstart', handleTouchStart, { passive: false });
    item.addEventListener('touchmove', handleTouchMove, { passive: false });
    item.addEventListener('touchend', handleTouchEnd);
  });
}

// Desktop drag handlers
function handleDragStart(e) {
  draggedElement = this;
  draggedIndex = parseInt(this.dataset.index);
  this.classList.add('dragging');
  e.dataTransfer.effectAllowed = 'move';
  e.dataTransfer.setData('text/html', this.innerHTML);
}

function handleDragOver(e) {
  e.preventDefault();
  e.dataTransfer.dropEffect = 'move';
  return false;
}

function handleDragEnter(e) {
  if (this !== draggedElement) {
    this.classList.add('drag-over');
  }
}

function handleDragLeave(e) {
  this.classList.remove('drag-over');
}

function handleDrop(e) {
  e.stopPropagation();
  e.preventDefault();
  
  const dropIndex = parseInt(this.dataset.index);
  
  if (draggedIndex !== dropIndex && draggedIndex !== null) {
    // Reorder array
    const movedExercise = selectedExercises.splice(draggedIndex, 1)[0];
    selectedExercises.splice(dropIndex, 0, movedExercise);
    
    // Update order property
    selectedExercises.forEach((ex, idx) => ex.order = idx);
    
    // Re-render
    renderWorkoutList();
    
    // Show feedback
    showNotification('Exercise reordered', 'info');
  }
  
  return false;
}

function handleDragEnd(e) {
  this.classList.remove('dragging');
  
  document.querySelectorAll('.exercise-item').forEach(item => {
    item.classList.remove('drag-over');
  });
  
  draggedIndex = null;
  draggedElement = null;
}

// Mobile touch handlers
let touchStartY = 0;
let touchStartX = 0;
let touchTimeout = null;
let isDraggingTouch = false;
let touchedElement = null;

function handleTouchStart(e) {
  touchedElement = this;
  touchStartY = e.touches[0].clientY;
  touchStartX = e.touches[0].clientX;
  
  // Long press to start drag
  touchTimeout = setTimeout(() => {
    isDraggingTouch = true;
    touchedElement.classList.add('dragging');
    navigator.vibrate && navigator.vibrate(50); // Haptic feedback
  }, 500);
}

function handleTouchMove(e) {
  if (!isDraggingTouch) {
    clearTimeout(touchTimeout);
    return;
  }
  
  e.preventDefault();
  
  const touch = e.touches[0];
  const touchY = touch.clientY;
  
  // Move element visually
  touchedElement.style.transform = `translateY(${touchY - touchStartY}px)`;
  
  // Detect element under touch
  const elementBelow = document.elementFromPoint(touch.clientX, touchY);
  const targetItem = elementBelow?.closest('.exercise-item');
  
  if (targetItem && targetItem !== touchedElement) {
    targetItem.classList.add('drag-over');
  }
}

function handleTouchEnd(e) {
  clearTimeout(touchTimeout);
  
  if (!isDraggingTouch) return;
  
  const touch = e.changedTouches[0];
  const elementBelow = document.elementFromPoint(touch.clientX, touch.clientY);
  const targetItem = elementBelow?.closest('.exercise-item');
  
  if (targetItem && targetItem !== touchedElement) {
    const fromIndex = parseInt(touchedElement.dataset.index);
    const toIndex = parseInt(targetItem.dataset.index);
    
    // Reorder
    const movedExercise = selectedExercises.splice(fromIndex, 1)[0];
    selectedExercises.splice(toIndex, 0, movedExercise);
    
    selectedExercises.forEach((ex, idx) => ex.order = idx);
  }
  
  // Reset
  touchedElement.style.transform = '';
  touchedElement.classList.remove('dragging');
  
  document.querySelectorAll('.exercise-item').forEach(item => {
    item.classList.remove('drag-over');
  });
  
  renderWorkoutList();
  
  isDraggingTouch = false;
  touchedElement = null;
}
```

```javascript
// Wake Lock variables
let wakeLock = null;
let isWorkoutActive = false;

// Request wake lock
async function requestWakeLock() {
  try {
    if ('wakeLock' in navigator) {
      wakeLock = await navigator.wakeLock.request('screen');
      console.log('✅ Wake Lock activated');
      
      wakeLock.addEventListener('release', () => {
        console.log('Wake Lock released');
        if (isWorkoutActive) {
          // Re-acquire if workout still active
          requestWakeLock();
        }
      });
    } else {
      console.warn('⚠️ Wake Lock API not supported');
      showNotification('Your browser does not support screen wake lock. Please keep your screen on manually.', 'warning');
    }
  } catch (err) {
    console.error('❌ Wake Lock error:', err);
  }
}

// Release wake lock
async function releaseWakeLock() {
  if (wakeLock) {
    try {
      await wakeLock.release();
      wakeLock = null;
      isWorkoutActive = false;
      console.log('✅ Wake Lock released');
    } catch (err) {
      console.error('❌ Error releasing wake lock:', err);
    }
  }
}

// Handle page visibility
document.addEventListener('visibilitychange', async () => {
  if (document.visibilityState === 'visible' && isWorkoutActive) {
    await requestWakeLock();
  }
});

// Start workout
function startWorkout() {
  if (selectedExercises.length === 0) {
    showNotification('❌ Please add exercises first', 'error');
    return;
  }
  
  // Activate wake lock

  // Reset all exercises

    ex.currentSet = 0;
    ex.completed = false;
  });
  
  // Switch to workout view
```

```javascript
  if (state === 'ready') {
    controls.innerHTML = `
      <button class="primary-btn full-width large" onclick="startSet()">
        Start Set ${currentSet}
      </button>
    `;
  } else if (state === 'active') {
    controls.innerHTML = `
      <button class="primary-btn full-width large" onclick="completeSet()">
        Complete Set
      </button>
    `;
  } else if (state === 'rest') {
    controls.innerHTML = `
      <button class="secondary-btn full-width" onclick="skipRest()">
        Skip Rest
      </button>
    `;
  }
}

// Start set
function startSet() {
  renderTimerDisplay('active');
  renderWorkoutControls('active');
}

// Complete set
function completeSet() {
  const exercise = selectedExercises[currentExerciseIndex];
  exercise.currentSet++;
  
  // Check if all sets completed for this exercise
  if (exercise.currentSet >= exercise.sets) {
    exercise.completed = true;
    
    // Move to next exercise
    const nextExerciseIndex = currentExerciseIndex + 1;
    
    if (nextExerciseIndex < selectedExercises.length) {
      // Start next exercise
      setTimeout(() => {
        startExercise(nextExerciseIndex);
      }, 1000);
    } else {
      // Workout complete
      completeWorkout();
    }
  } else {
    // Start rest timer
    startRestTimer(exercise.rest);
  }
}

// Start rest timer
function startRestTimer(duration) {
  timeRemaining = duration;
  isResting = true;
  
  renderTimerDisplay('rest');
  renderWorkoutControls('rest');
  
  timerInterval = setInterval(() => {
    timeRemaining--;
    
    document.getElementById('timer-value').textContent = formatTime(timeRemaining);
    
    if (timeRemaining <= 0) {
      clearInterval(timerInterval);
      isResting = false;
      
      // Vibrate and notify
      navigator.vibrate && navigator.vibrate([200, 100, 200]);
      showNotification('Rest complete! Ready for next set', 'info');
      
      // Start next set
      startExercise(currentExerciseIndex);
    }
  }, 1000);
}

// Skip rest
function skipRest() {
  if (timerInterval) {
    clearInterval(timerInterval);
    timerInterval = null;
  }
  
  isResting = false;
  startExercise(currentExerciseIndex);
}

// Format time (seconds to MM:SS)
function formatTime(seconds) {
  const mins = Math.floor(seconds / 60);
  const secs = seconds % 60;
  return `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
}

// Show completion screen
function showCompletionScreen() {
  const container = document.getElementById('workout-container');
  
  const totalSets = selectedExercises.reduce((sum, ex) => sum + ex.sets, 0);
  const totalExercises = selectedExercises.length;
  
  container.innerHTML = `
    <div class="completion-screen">
      <div class="completion-icon">🎉</div>
      <h1>Workout Complete!</h1>
      <div class="completion-stats">
        <div class="completion-stat">
          <div class="stat-value">${totalExercises}</div>
          <div class="stat-label">Exercises</div>
        </div>
        <div class="completion-stat">
          <div class="stat-value">${totalSets}</div>
          <div class="stat-label">Sets</div>
        </div>
      </div>
      <button class="primary-btn full-width" onclick="resetWorkout()">
        Done
      </button>
    </div>
  `;
}

// Reset workout
function resetWorkout() {
  selectedExercises.forEach(ex => {
    ex.currentSet = 0;
    ex.completed = false;
  });
  
  showWorkoutBuilder();
}

// Show workout builder
function showWorkoutBuilder() {
  location.reload(); // Simple reset - reload the page
}
```

```javascript
// Show notification
function showNotification(message, type = 'info') {
  // Remove existing notifications
  const existing = document.querySelector('.notification');
  if (existing) {
    existing.remove();
  }
  
  const notification = document.createElement('div');
  notification.className = `notification notification-${type}`;
  notification.textContent = message;
  
  document.body.appendChild(notification);
  
  // Animate in
  setTimeout(() => {
    notification.classList.add('show');
  }, 10);
  
  // Auto remove
  setTimeout(() => {
    notification.classList.remove('show');
    setTimeout(() => {
      notification.remove();
    }, 300);
  }, 3000);
}
```
