import { useEffect, useState } from 'react'
import { supabase } from './lib/supabase'

export function TestConnection() {
  const [status, setStatus] = useState('בודק חיבור...')
  const [workouts, setWorkouts] = useState<any[]>([])

  useEffect(() => {
    testConnection()
  }, [])

  const testConnection = async () => {
    try {
      console.log('🔍 בודק חיבור ל-Supabase...')
      
      const { data, error } = await supabase
        .from('workout_templates')
        .select('*')
        .limit(3)
      
      if (error) {
        console.error('❌ שגיאה:', error)
        setStatus('❌ שגיאה בחיבור: ' + error.message)
        return
      }
      
      console.log('✅ הצלחה! נמצאו אימונים:', data)
      setStatus('✅ חיבור תקין!')
      setWorkouts(data || [])
      
    } catch (err: any) {
      console.error('❌ שגיאה לא צפויה:', err)
      setStatus('❌ שגיאה: ' + err.message)
    }
  }

  return (
    <div style={{ 
      padding: '40px', 
      maxWidth: '800px', 
      margin: '0 auto',
      fontFamily: 'Arial'
    }}>
      <h1 style={{ fontSize: '32px', marginBottom: '20px' }}>
        בדיקת חיבור Supabase
      </h1>
      
      <div style={{ 
        padding: '20px', 
        background: '#f0f0f0', 
        borderRadius: '8px',
        marginBottom: '20px'
      }}>
        <h2 style={{ fontSize: '24px' }}>{status}</h2>
      </div>
      
      {workouts.length > 0 && (
        <div>
          <h3 style={{ fontSize: '20px', marginBottom: '10px' }}>
            אימונים מהמסד נתונים:
          </h3>
          {workouts.map((workout) => (
            <div 
              key={workout.id}
              style={{ 
                padding: '15px', 
                background: 'white', 
                border: '1px solid #ddd',
                borderRadius: '8px',
                marginBottom: '10px'
              }}
            >
              <div style={{ fontWeight: 'bold', fontSize: '18px' }}>
                יום {workout.day_number}: {workout.title}
              </div>
              <div style={{ color: '#666', marginTop: '5px' }}>
                סוג: {workout.workout_type}
              </div>
            </div>
          ))}
        </div>
      )}
      
      <div style={{ marginTop: '30px', color: '#666' }}>
        פתח את הקונסול (F12) לפרטים נוספים
      </div>
    </div>
  )
}
